const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; let recognition = null; let isLiveVoiceActive = false; if (SpeechRecognition) { recognition = new SpeechRecognition(); recognition.continuous = true; // Arxa-arxaya dinləməyə davam etsin recognition.interimResults = false; recognition.lang = 'en-US'; recognition.onresult = (event) => { const lastIndex = event.results.length - 1; const transcript = event.results[lastIndex][0].transcript.trim(); if (transcript) { document.getElementById('user-input').value = transcript; sendMessage(); } }; recognition.onerror = (event) => { console.error("Speech Recognition Error:", event.error); if (event.error === 'not-allowed') { alert("Please allow microphone access in your browser settings!"); } }; recognition.onend = () => { // Əgər rejim aktivdirsə və səs kəsilibsə, avtomatik yenidən dinləməyə başla if (isLiveVoiceActive) { try { recognition.start(); } catch(e) {} } }; } else { alert("Web Speech API is not supported in this browser. Please use Google Chrome."); } function toggleLiveVoice() { const btn = document.getElementById('btn-live-voice'); const indicator = document.getElementById('live-indicator'); if (!recognition) return; if (!isLiveVoiceActive) { isLiveVoiceActive = true; try { recognition.start(); btn.classList.add('active'); indicator.style.display = 'block'; } catch (err) { console.error(err); } } else { isLiveVoiceActive = false; recognition.stop(); btn.classList.remove('active'); indicator.style.display = 'none'; } }
