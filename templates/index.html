// Səsli Danışma (Brauzerin dili və ya ümumi dil dəstəyi ilə) let recognition; function toggleVoice() { let micBtn = document.getElementById("mic-btn"); if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) { alert("Brauzeriniz səs tanıma funksiyasını dəstəkləmir."); return; } const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (recognition && micBtn.classList.contains("recording")) { recognition.stop(); micBtn.classList.remove("recording"); return; } recognition = new SpeechRecognition(); // Dil təyin etmirik ki, brauzer istifadəçinin danışdığı dili avtomatik tutsun (və ya brauzer dilinə uyğunlaşsın) recognition.start(); micBtn.classList.add("recording"); recognition.onresult = (event) => { let transcript = event.results[0][0].transcript; document.getElementById("user-input").value = transcript; micBtn.classList.remove("recording"); sendMessage(); }; recognition.onerror = () => micBtn.classList.remove("recording"); recognition.onend = () => micBtn.classList.remove("recording"); } // Mətni Səsləndirmə (Avtomatik dil tənzimlənməsi) function speakText(text) { let synth = window.speechSynthesis; let utterance = new SpeechSynthesisUtterance(text); // Avtomatik olaraq istifadəçinin sistem/səs dilinə uyğunlaşır synth.speak(utterance); }
